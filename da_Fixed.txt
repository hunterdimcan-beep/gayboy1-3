local function camera()
	return workspace.CurrentCamera
end

local function character(player)
	return player and player.Character
end

local function humanoid(player)
	local c = character(player)
	return c and c:FindFirstChildOfClass("Humanoid")
end

local function root(player)
	local c = character(player)
	return c and c:FindFirstChild("HumanoidRootPart")
end

local function head(player)
	local c = character(player)
	return c and c:FindFirstChild("Head")
end

-- Aim point part for CamLock (Head / Chest / LowerChest)
local function getAimPart(player)
	local c = character(player)
	if not c then
		return nil
	end

	local point = Config.Combat.AimPoint or "Head"

	if point == "Chest" then
		return c:FindFirstChild("UpperTorso")
			or c:FindFirstChild("Torso")
			or c:FindFirstChild("Head")
	elseif point == "LowerChest" then
		return c:FindFirstChild("LowerTorso")
			or c:FindFirstChild("Torso")
			or c:FindFirstChild("UpperTorso")
			or c:FindFirstChild("Head")
	else
		-- Head (default)
		return c:FindFirstChild("Head")
			or c:FindFirstChild("UpperTorso")
			or c:FindFirstChild("Torso")
	end
end

local function alive(player)
	local h = humanoid(player)
	return h and h.Health > 0
end

------------------------------------------------------------
-- DOWNED CHECK
------------------------------------------------------------

local function isDowned(player)
	local c = character(player)
	if not c then
		return false
	end

	local names = {
		"Downed",
		"IsDowned",
		"Knocked",
		"KnockedDown",
		"Ragdolled",
		"Ragdoll",
	}

	for _, name in ipairs(names) do
		if c:GetAttribute(name) == true then
			return true
		end

		local object = c:FindFirstChild(name)

		if object and object:IsA("BoolValue") and object.Value then
			return true
		end
	end

	local bodyEffects = c:FindFirstChild("BodyEffects")

	if bodyEffects then
		for _, name in ipairs({
			"Downed",
			"K.O",
			"KO",
			"Knocked",
		}) do
			local object = bodyEffects:FindFirstChild(name)

			if object then
				if object:IsA("BoolValue") and object.Value then
					return true
				end

				if object:IsA("NumberValue") and object.Value ~= 0 then
					return true
				end
			end
		end
	end

	return false
end

local function friend(player)
	local ok, result = pcall(function()
		return LocalPlayer:IsFriendsWith(player.UserId)
	end)

	return ok and result
end

local function sameTeam(player)
	if not Config.Combat.TeamCheck then
		return false
	end

	return LocalPlayer.Team ~= nil
		and player.Team ~= nil
		and LocalPlayer.Team == player.Team
end

local function visible(player)
	if not Config.Combat.WallCheck then
		return true
	end

	local h = head(player)
	local cam = camera()

	if not h or not cam then
		return false
	end

	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = {
		LocalPlayer.Character
	}
	params.IgnoreWater = true

	local origin = cam.CFrame.Position
	local direction = h.Position - origin

	local result = workspace:Raycast(
		origin,
		direction,
		params
	)

	if not result then
		return true
	end

	return result.Instance:IsDescendantOf(player.Character)
end

local function validTarget(player)
	if not player or player == LocalPlayer then
		return false
	end

	if not alive(player) then
		return false
	end

	if Config.Combat.IgnoreDowned and isDowned(player) then
		return false
	end

	if Config.Combat.IgnoreFriends and friend(player) then
		return false
	end

	if sameTeam(player) then
		return false
	end

	if not head(player) then
		return false
	end

	return visible(player)
end

local function getTarget()
	local cam = camera()
	if not cam then
		return nil
	end

	local viewport = cam.ViewportSize

	local center = Vector2.new(
		viewport.X / 2,
		viewport.Y / 2
	)

	local best = nil
	local bestDistance = Config.Combat.AimFOV

	for _, player in ipairs(Players:GetPlayers()) do
		if validTarget(player) then
			local aim = getAimPart(player)
			if aim then
				local position, onScreen =
					cam:WorldToViewportPoint(aim.Position)

				if onScreen and position.Z > 0 then
					local screenPosition = Vector2.new(
						position.X,
						position.Y
					)

					local distance =
						(screenPosition - center).Magnitude

					if distance < bestDistance then
						bestDistance = distance
						best = player
					end
				end
			end
		end
	end

	return best
end

local function makeMainDraggable(handle)
	local dragging = false
	local start
	local original

	handle.InputBegan:Connect(function(input)
		if input.UserInputType ==
			Enum.UserInputType.MouseButton1
			or input.UserInputType ==
			Enum.UserInputType.Touch then

			dragging = true
			start = input.Position
			original = Main.Position

			input.Changed:Connect(function()
				if input.UserInputState ==
					Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if not dragging then
			return
		end

		if input.UserInputType ==
			Enum.UserInputType.MouseMovement
			or input.UserInputType ==
			Enum.UserInputType.Touch then

			local delta =
				input.Position - start

			Main.Position = UDim2.new(
				original.X.Scale,
				original.X.Offset + delta.X,
				original.Y.Scale,
				original.Y.Offset + delta.Y
			)
		end
	end)
end

		State.CurrentTab = name

		for tab, object in pairs(TabButtons) do
			local selected = tab == name

			object.BackgroundColor3 =
				selected
				and Config.Settings.UIColor
				or Color3.fromRGB(25, 25, 32)

			object.TextColor3 =
				selected

				or Color3.fromRGB(175, 175, 185)
		end

		for pageName, page in pairs(PageObjects) do
			page.Visible = pageName == name
		end
	end)

	indicator.Size = UDim2.fromOffset(18, 18)
	indicator.Position = UDim2.fromOffset(2, 2)
	indicator.BackgroundColor3 = Color3.fromRGB(145, 145, 155)
	indicator.BorderSizePixel = 0

	indicatorCorner.CornerRadius = UDim.new(1, 0)
	indicatorCorner.Parent = indicator

	local value = false

	local function setValue(v, invoke)
		value = v == true

		if value then
			button.BackgroundColor3 =
				Config.Settings.UIColor
			indicator.BackgroundColor3 =
				Color3.fromRGB(255,255,255)
			indicator.Position =
				UDim2.new(1,-20,0,2)
		else
			button.BackgroundColor3 =
				Color3.fromRGB(40,40,48)
			indicator.BackgroundColor3 =
				Color3.fromRGB(145,145,155)
			indicator.Position =
				UDim2.fromOffset(2,2)
		end

		if invoke and callback then
			callback(value)
		end
	end

	button.Activated:Connect(function()
		setValue(not value, true)
	end)

	Controls[path] = function(v)
		setValue(v, true)
	end

	local r, k = string.match(path, "^(.-)%.([^%.]+)$")
	if r and k and Config[r] then
		setValue(Config[r][k], false)
	end
end

	fill.Size = UDim2.new(0,0,1,0)
	fill.BackgroundColor3 = Config.Settings.UIColor
	fill.BorderSizePixel = 0
	fill.Parent = bar

	local fillCorner = Instance.new("UICorner")
	fillCorner.CornerRadius = UDim.new(1,0)
	fillCorner.Parent = fill

	table.insert(AccentObjects, {
		Object = fill,
		Type = "Background"
	})

	local knob = Instance.new("Frame")
	knob.Size = UDim2.fromOffset(14,14)
	knob.AnchorPoint = Vector2.new(.5,.5)
	knob.Position = UDim2.new(0,0,.5,0)
	knob.BackgroundColor3 = Config.Settings.UIColor
	knob.BorderSizePixel = 0
	knob.Active = true
	knob.Parent = bar

	local knobCorner = Instance.new("UICorner")
	knobCorner.CornerRadius = UDim.new(1,0)
	knobCorner.Parent = knob

	table.insert(AccentObjects, {
		Object = knob,
		Type = "Background"
	})

	local value = minValue
	local dragging = false

	local function setValue(v, invoke)
		value = math.clamp(
			tonumber(v) or minValue,
			minValue,
			maxValue
		)

		local percent =
			(value-minValue) /
			(maxValue-minValue)

		fill.Size =
			UDim2.new(percent,0,1,0)

		knob.Position =
			UDim2.new(percent,0,.5,0)

		valueLabel.Text =
			tostring(math.round(value))

		if invoke and callback then
			callback(value)
		end
	end

	local function update(x)
		local left = bar.AbsolutePosition.X
		local width = bar.AbsoluteSize.X

		local percent =
			math.clamp(
				(x-left)/width,
				0,
				1
			)

		setValue(
			minValue +
			(maxValue-minValue)*percent,
			true
		)
	end

	local function begin(input)
		dragging = true
		update(input.Position.X)
	end

	bar.InputBegan:Connect(function(input)
		if input.UserInputType ==
			Enum.UserInputType.MouseButton1
			or input.UserInputType ==
			Enum.UserInputType.Touch then
			begin(input)
		end
	end)

	knob.InputBegan:Connect(function(input)
		if input.UserInputType ==
			Enum.UserInputType.MouseButton1
			or input.UserInputType ==
			Enum.UserInputType.Touch then
			begin(input)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if dragging and (
			input.UserInputType ==
				Enum.UserInputType.MouseMovement
			or input.UserInputType ==
				Enum.UserInputType.Touch
		) then
			update(input.Position.X)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType ==
			Enum.UserInputType.MouseButton1
			or input.UserInputType ==
			Enum.UserInputType.Touch then
			dragging = false
		end
	end)

	Controls[path] = function(v)
		setValue(v, true)
	end

	local r,k = string.match(path, "^(.-)%.([^%.]+)$")
	if r and k and Config[r] then
		setValue(Config[r][k], false)
	end
end

-- FLOATING BUTTON DRAG
------------------------------------------------------------

local function makeButtonDraggable(button, clickFunction)
	local dragging = false
	local moved = false
	local start
	local original
	local activeInput = nil -- only track ONE finger/mouse so extra touches don't move the button
	local DRAG_THRESHOLD = 30

	button.InputBegan:Connect(function(input)
		if dragging then
			return -- ignore extra fingers
		end

		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			dragging = true
			moved = false
			start = input.Position
			original = button.Position
			activeInput = input

			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					if activeInput == input then
						dragging = false
						activeInput = nil
						if not moved then
							clickFunction()
						end
					end
				end
			end)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if not dragging or input ~= activeInput then
			return -- only the finger that started the drag
		end

		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then

			local delta = input.Position - start

			if not moved then
				if delta.Magnitude < DRAG_THRESHOLD then
					return
				end
				moved = true
			end

			button.Position = UDim2.new(
				original.X.Scale,
				original.X.Offset + delta.X,
				original.Y.Scale,
				original.Y.Offset + delta.Y
			)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input == activeInput then
			dragging = false
			activeInput = nil
		end
	end)
end

makeButtonDraggable(
	LockButton,
	function()
		-- Temporary pause only — menu Cam Lock toggle stays on, FOV stays
		State.CamLockActive = not State.CamLockActive
		if not State.CamLockActive then
			State.CamTarget = nil
		end
	end
)

makeButtonDraggable(
	FlyButton,
	function()
		-- Temporary pause only — menu Fly toggle stays on
		-- updateFly() will start/stop BodyVelocity based on State.FlyPaused
		State.FlyPaused = not State.FlyPaused
	end
)

local function applyPreset(preset)
	if preset == "Rage" then
		Config.Combat.CamLock = true
		Config.Combat.AimFOV = 400
		Config.Combat.CamLockSmoothing = 3
		Config.Combat.AimPoint = "Chest"
		Config.Combat.WallCheck = false
		Config.Combat.TeamCheck = false
		Config.Combat.IgnoreDowned = false
		Config.Combat.IgnoreFriends = false
		Config.Movement.SpeedEnabled = true
		Config.Movement.Speed = 28
		Config.Movement.JumpEnabled = true
		Config.Movement.JumpPower = 70
		Config.ESP.Skeleton = true
		Config.ESP.Box = true
		Config.ESP.Chams = true
		Config.ESP.Names = true
		Config.ESP.Tracers = true
	elseif preset == "Legit" then
		Config.Combat.CamLock = true
		Config.Combat.AimFOV = 80
		Config.Combat.CamLockSmoothing = 22
		Config.Combat.AimPoint = "Head"
		Config.Combat.WallCheck = true
		Config.Combat.TeamCheck = true
		Config.Combat.IgnoreDowned = true
		Config.Combat.IgnoreFriends = true
		Config.Movement.SpeedEnabled = false
		Config.Movement.Speed = 16
		Config.Movement.JumpEnabled = false
		Config.Movement.JumpPower = 50
		Config.Movement.Fly = false
		Config.ESP.Skeleton = true
		Config.ESP.Box = false
		Config.ESP.Chams = false
		Config.ESP.Names = true
		Config.ESP.Tracers = false
		Config.ESP.Corners = false
		Config.ESP.AimChecker = false
	end

	-- Refresh all UI controls to match config
	for path, setter in pairs(Controls) do
		local r, k = string.match(path, "^(.-)%.([^%.]+)$")
		if r and k and Config[r] and Config[r][k] ~= nil then
			setter(Config[r][k])
		end
	end

	State.CamLockActive = true
	State.FlyPaused = false
	AimPointLabel.Text = "Current: " .. (Config.Combat.AimPoint or "Head")

	if ConfigStatus then
		ConfigStatus.Text = "Loaded preset: " .. preset
	end
end

local function performanceMode(enabled)
	if enabled then
		Lighting.GlobalShadows = false

		for _, obj in ipairs(workspace:GetDescendants()) do
			if obj:IsA("ParticleEmitter")
				or obj:IsA("Trail")
				or obj:IsA("Smoke")
				or obj:IsA("Fire")
				or obj:IsA("Sparkles") then

				obj.Enabled = false
			end
		end
	end
end

destroyESP = function(player)
	local data = ESPObjects[player]
	if not data then
		return
	end

	if data.Name then
		data.Name:Destroy()
	end

	if data.Tracer then
		data.Tracer:Destroy()
	end

	if data.Skeleton then
		for _, item in ipairs(data.Skeleton) do
			if item then
				if item.Frame then
					item.Frame:Destroy()
				end
				if item.Line then
					item.Line:Destroy()
				end
				if item.Beam then
					item.Beam:Destroy()
				end
				if item.A0 then
					item.A0:Destroy()
				end
				if item.A1 then
					item.A1:Destroy()
				end
			end
		end
	end

	if data.Boxes then
		for _, box in ipairs(data.Boxes) do
			if box then
				box:Destroy()
			end
		end
	end

	if data.Chams then
		data.Chams:Destroy()
	end

	if data.Corners then
		for _, f in ipairs(data.Corners) do
			if f then
				f:Destroy()
			end
		end
	end

	if data.Aim then
		data.Aim:Destroy()
	end

	ESPObjects[player] = nil
end

-- Permanent folder so adornments survive camera changes
local ESPFolder = Instance.new("Folder")
ESPFolder.Name = "StarbornESP"
ESPFolder.Parent = workspace

local function getAdornee()
	local t = workspace:FindFirstChildOfClass("Terrain")
	if t then
		return t
	end
	local p = workspace:FindFirstChild("StarbornESPAdornee")
	if not p then
		p = Instance.new("Part")
		p.Name = "StarbornESPAdornee"
		p.Anchored = true
		p.CanCollide = false
		p.Transparency = 1
		p.Size = Vector3.new(1, 1, 1)
		p.Position = Vector3.zero
		p.Parent = workspace
	end
	return p
end

local function espAlpha()
	-- slider is 0-90, convert to 0-0.9 transparency
	return math.clamp((Config.Settings.ESPTransparency or 0) / 100, 0, 0.9)
end

local function espThick()
	return math.clamp(Config.Settings.ESPThickness or 3, 1, 10)
end

local function makeLine()
	local line = Instance.new("LineHandleAdornment")
	line.Thickness = espThick()
	line.Transparency = espAlpha()
	line.AlwaysOnTop = true
	line.Visible = false
	line.ZIndex = 10
	line.Adornee = getAdornee()
	line.Parent = ESPFolder
	return line
end

-- Straight beam between two parts (guaranteed straight lines)
local function makeBeam(part0, part1)
	local a0 = Instance.new("Attachment")
	a0.Name = "StarbornA0"
	a0.Parent = part0

	local a1 = Instance.new("Attachment")
	a1.Name = "StarbornA1"
	a1.Parent = part1

	local beam = Instance.new("Beam")
	beam.Name = "StarbornBeam"
	beam.Attachment0 = a0
	beam.Attachment1 = a1
	local w = math.clamp(espThick() * 0.06, 0.06, 0.5)
	beam.Width0 = w
	beam.Width1 = w
	beam.FaceCamera = true
	beam.LightEmission = 1
	beam.LightInfluence = 0
	beam.Color = ColorSequence.new(Config.Settings.ESPColor)
	beam.Transparency = NumberSequence.new(espAlpha())
	beam.Segments = 1 -- single segment = perfectly straight
	beam.Parent = part0

	return { Beam = beam, A0 = a0, A1 = a1 }
end

		Name = nil,

		Aim = nil,

local function updateLine(line, a, b)
	if not line then
		return
	end

	if not line.Adornee or not line.Adornee.Parent then
		line.Adornee = getAdornee()
	end

	if not a or not b then
		line.Visible = false
		return
	end

	local delta = b - a
	local length = delta.Magnitude

	if length <= 0.05 then
		line.Visible = false
		return
	end

	local mid = (a + b) * 0.5
	line.CFrame = CFrame.lookAt(mid, b) * CFrame.Angles(math.rad(90), 0, 0)
	line.Length = length
	line.Thickness = 2
	line.Color3 = Config.Settings.ESPColor
	line.AlwaysOnTop = true
	line.Visible = true
end

local function updateESP(player)
	local c = character(player)

	if not c then
		destroyESP(player)
		return
	end

	local data = ESPObjects[player]

	if not data or data.Character ~= c then
		setupESP(player)
		data = ESPObjects[player]
	end

	if not data then
		return
	end

	local isAlive = alive(player)

	--------------------------------------------------------
	-- SKELETON (screen-space stick figure like the reference image)
	--------------------------------------------------------
	if Config.ESP.Skeleton then
		local bones = c:FindFirstChild("UpperTorso") and R15 or R6

		if #data.Skeleton ~= #bones then
			setupESP(player)
			data = ESPObjects[player]
		end

		local cam = camera()
		local thick = math.max(1, math.floor(espThick()))
		local alpha = espAlpha()
		local col = Config.Settings.ESPColor

		if data and cam then
			for i, bone in ipairs(bones) do
				local item = data.Skeleton[i]
				if item and item.Frame then
					local partA = c:FindFirstChild(bone[1])
					local partB = c:FindFirstChild(bone[2])

					if partA and partB and isAlive then
						local p1, on1 = cam:WorldToViewportPoint(partA.Position)
						local p2, on2 = cam:WorldToViewportPoint(partB.Position)

						-- draw if at least one end is in front of camera
						if p1.Z > 0 and p2.Z > 0 then
							local from = Vector2.new(p1.X, p1.Y)
							local to = Vector2.new(p2.X, p2.Y)
							local delta = to - from
							local length = delta.Magnitude

							if length > 1 then
								item.Frame.Position = UDim2.fromOffset(
									(from.X + to.X) * 0.5,
									(from.Y + to.Y) * 0.5
								)
								item.Frame.Size = UDim2.fromOffset(length, thick)
								item.Frame.Rotation = math.deg(math.atan2(delta.Y, delta.X))
								item.Frame.BackgroundColor3 = col
								item.Frame.BackgroundTransparency = alpha
								item.Frame.Visible = true
							else
								item.Frame.Visible = false
							end
						else
							item.Frame.Visible = false
						end
					else
						item.Frame.Visible = false
					end
				end
			end
		end
	else
		for _, item in ipairs(data.Skeleton) do
			if item and item.Frame then
				item.Frame.Visible = false
			end
		end
	end

	--------------------------------------------------------
	-- BOX
	--------------------------------------------------------
	if Config.ESP.Box and data.Boxes then
		local alpha = espAlpha()
		for _, box in ipairs(data.Boxes) do
			if box and box.Parent then
				box.Visible = isAlive
				box.Color3 = Color3.fromRGB(255, 50, 50)
				box.Transparency = alpha
				box.AlwaysOnTop = true
				if box.Adornee and box.Adornee:IsA("BasePart") then
					box.Size = box.Adornee.Size * 1.08
				end
			end
		end
	elseif data.Boxes then
		for _, box in ipairs(data.Boxes) do
			if box then
				box.Visible = false
			end
		end
	end

	--------------------------------------------------------
	-- CHAMS
	--------------------------------------------------------
	if data.Chams then
		data.Chams.Enabled = Config.ESP.Chams and isAlive
		if Config.ESP.Chams then
			data.Chams.FillColor = Color3.fromRGB(255, 40, 40)
			data.Chams.OutlineColor = Color3.fromRGB(255, 80, 80)
			data.Chams.FillTransparency = math.clamp(espAlpha(), 0.15, 0.7)
			data.Chams.OutlineTransparency = 0
			data.Chams.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		end
	end

	--------------------------------------------------------
	-- CORNER BOXES (2D)
	--------------------------------------------------------
	if Config.ESP.Corners and data.Corners and #data.Corners >= 8 then
		local hrp = root(player)
		local cam = camera()
		if hrp and cam and isAlive then
			-- Get character bounds in screen space
			local cf = hrp.CFrame
			local size = Vector3.new(4, 6, 2)
			local hum = humanoid(player)
			if hum then
				size = Vector3.new(4, hum.HipHeight * 2 + 2, 2)
			end

			-- sample 8 corners of a bounding box around the character
			local half = size * 0.5
			local corners3d = {
				cf * Vector3.new(-half.X,  half.Y, -half.Z),
				cf * Vector3.new( half.X,  half.Y, -half.Z),
				cf * Vector3.new(-half.X, -half.Y, -half.Z),
				cf * Vector3.new( half.X, -half.Y, -half.Z),
				cf * Vector3.new(-half.X,  half.Y,  half.Z),
				cf * Vector3.new( half.X,  half.Y,  half.Z),
				cf * Vector3.new(-half.X, -half.Y,  half.Z),
				cf * Vector3.new( half.X, -half.Y,  half.Z),
			}

			local minX, minY = math.huge, math.huge
			local maxX, maxY = -math.huge, -math.huge
			local onScreen = false

			for _, worldPos in ipairs(corners3d) do
				local sp, vis = cam:WorldToViewportPoint(worldPos)
				if vis and sp.Z > 0 then
					onScreen = true
					minX = math.min(minX, sp.X)
					minY = math.min(minY, sp.Y)
					maxX = math.max(maxX, sp.X)
					maxY = math.max(maxY, sp.Y)
				end
			end

			if onScreen then
				local thick = 2
				local len = math.clamp((maxX - minX) * 0.25, 8, 30)
				local col = Config.Settings.ESPColor

				-- 4 corners × 2 lines each = 8 frames
				-- TL horizontal, TL vertical
				-- TR horizontal, TR vertical
				-- BL horizontal, BL vertical
				-- BR horizontal, BR vertical
				local specs = {
					{minX, minY, len, thick},           -- TL H
					{minX, minY, thick, len},           -- TL V
					{maxX - len, minY, len, thick},     -- TR H
					{maxX - thick, minY, thick, len},   -- TR V
					{minX, maxY - thick, len, thick},   -- BL H
					{minX, maxY - len, thick, len},     -- BL V
					{maxX - len, maxY - thick, len, thick}, -- BR H
					{maxX - thick, maxY - len, thick, len}, -- BR V
				}

				for i, s in ipairs(specs) do
					local f = data.Corners[i]
					f.Position = UDim2.fromOffset(s[1], s[2])
					f.Size = UDim2.fromOffset(s[3], s[4])
					f.BackgroundColor3 = col
					f.Visible = true
				end
			else
				for _, f in ipairs(data.Corners) do
					f.Visible = false
				end
			end
		else
			for _, f in ipairs(data.Corners) do
				f.Visible = false
			end
		end
	elseif data.Corners then
		for _, f in ipairs(data.Corners) do
			f.Visible = false
		end
	end

	--------------------------------------------------------
	-- NAME
	--------------------------------------------------------
	if data.Name then
		data.Name.Enabled = Config.ESP.Names and isAlive
		local label = data.Name:FindFirstChildOfClass("TextLabel")
		if label then
			label.TextColor3 = Config.Settings.ESPColor
		end
	end

	--------------------------------------------------------
	-- TRACER
	--------------------------------------------------------
	if Config.ESP.Tracers and data.Tracer then
		local h = head(player)
		local cam = camera()
		if h and cam and isAlive then
			local pos, onScreen = cam:WorldToViewportPoint(h.Position)
			if onScreen and pos.Z > 0 then
				local viewport = cam.ViewportSize
				local start = Vector2.new(viewport.X / 2, viewport.Y - 15)
				local finish = Vector2.new(pos.X, pos.Y)
				local delta = finish - start

				data.Tracer.Position = UDim2.fromOffset(
					(start.X + finish.X) / 2,
					(start.Y + finish.Y) / 2
				)
				data.Tracer.Size = UDim2.fromOffset(delta.Magnitude, 1)
				data.Tracer.Rotation = math.deg(math.atan2(delta.Y, delta.X))
				data.Tracer.BackgroundColor3 = Config.Settings.ESPColor
				data.Tracer.Visible = true
			else
				data.Tracer.Visible = false
			end
		else
			data.Tracer.Visible = false
		end
	elseif data.Tracer then
		data.Tracer.Visible = false
	end

	--------------------------------------------------------
	-- AIM CHECKER (where they are looking – head → look direction)
	--------------------------------------------------------
	if Config.ESP.AimChecker and data.Aim then
		local h = head(player)
		local cam = camera()
		if h and cam and isAlive then
			-- Use head look, fallback to root facing
			local look = h.CFrame.LookVector
			local r = root(player)
			if r then
				-- blend with body facing so it stays sensible on R6
				look = (h.CFrame.LookVector + r.CFrame.LookVector * 0.35).Unit
			end

			local origin = h.Position
			local finish = origin + look * 40

			local p1, on1 = cam:WorldToViewportPoint(origin)
			local p2, on2 = cam:WorldToViewportPoint(finish)

			if p1.Z > 0 and p2.Z > 0 then
				local from = Vector2.new(p1.X, p1.Y)
				local to = Vector2.new(p2.X, p2.Y)
				local delta = to - from
				local length = delta.Magnitude
				local thick = math.max(2, math.floor(espThick()))

				if length > 2 then
					data.Aim.Position = UDim2.fromOffset(
						(from.X + to.X) * 0.5,
						(from.Y + to.Y) * 0.5
					)
					data.Aim.Size = UDim2.fromOffset(length, thick)
					data.Aim.Rotation = math.deg(math.atan2(delta.Y, delta.X))
					data.Aim.BackgroundColor3 = Config.Settings.AimCheckerColor
					data.Aim.BackgroundTransparency = espAlpha()
					data.Aim.Visible = true
				else
					data.Aim.Visible = false
				end
			else
				data.Aim.Visible = false
			end
		else
			data.Aim.Visible = false
		end
	elseif data.Aim then
		data.Aim.Visible = false
	end
end

local originalHipHeight = {}

local function updateMovement()
	local c = LocalPlayer.Character
	if not c then
		return
	end

	local h = c:FindFirstChildOfClass("Humanoid")
	local r = c:FindFirstChild("HumanoidRootPart")

	if not h then
		return
	end

	if originalHipHeight[c] == nil then
		originalHipHeight[c] = h.HipHeight
	end

	local speed = 16

	if Config.Movement.SpeedEnabled then
		speed = Config.Movement.Speed
	end

	if Config.Movement.Sprint then
		speed = math.max(speed,24)
	end

	if Config.Movement.Crouch then
		speed = math.min(speed,8)
	end

	h.WalkSpeed = speed

	if Config.Movement.JumpEnabled then
		h.UseJumpPower = true
		h.JumpPower = Config.Movement.JumpPower
	end

	if Config.Movement.Crouch then
		h.HipHeight =
			originalHipHeight[c] * .55
	else
		h.HipHeight =
			originalHipHeight[c]
	end

	if Config.Movement.SpinBot and r then
		r.CFrame =
			r.CFrame *
			CFrame.Angles(
				0,
				math.rad(Config.Movement.SpinSpeed),
				0
			)
	end
end

stopFly = function()
	State.FlyActive = false
	-- Do NOT hide FlyButton — it stays visible so user can turn fly back on

	if State.FlyBV then
		State.FlyBV:Destroy()
		State.FlyBV = nil
	end
	if State.FlyBG then
		State.FlyBG:Destroy()
		State.FlyBG = nil
	end

	local c = LocalPlayer.Character
	if c then
		local h = c:FindFirstChildOfClass("Humanoid")
		if h then
			h.PlatformStand = false
		end
	end
end

startFly = function()
	local c = LocalPlayer.Character
	if not c then
		return
	end

	local h = c:FindFirstChildOfClass("Humanoid")
	local r = c:FindFirstChild("HumanoidRootPart")
	if not h or not r then
		return
	end

	-- Already running
	if State.FlyBV and State.FlyBV.Parent == r then
		return
	end

	stopFly() -- clean any leftover

	-- Lazy-load ControlModule (works on mobile + PC)
	if not State.FlyControlModule then
		local ok, mod = pcall(function()
			return require(
				LocalPlayer:WaitForChild("PlayerScripts")
					:WaitForChild("PlayerModule")
					:WaitForChild("ControlModule")
			)
		end)
		if ok then
			State.FlyControlModule = mod
		end
	end

	local bv = Instance.new("BodyVelocity")
	bv.Name = "StarbornFlyBV"
	bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
	bv.Velocity = Vector3.zero
	bv.Parent = r

	local bg = Instance.new("BodyGyro")
	bg.Name = "StarbornFlyBG"
	bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
	bg.P = 10000
	bg.Parent = r

	State.FlyBV = bv
	State.FlyBG = bg
	State.FlyActive = true

	h.PlatformStand = true
end

local function updateFly()
	-- Menu toggle off → fully stop
	if not Config.Movement.Fly then
		if State.FlyActive or State.FlyBV then
			stopFly()
		end
		return
	end

	-- Floating button temporary pause → stop physics but keep menu toggle on
	if State.FlyPaused then
		if State.FlyBV then
			stopFly()
		end
		return
	end

	local c = LocalPlayer.Character
	if not c then
		stopFly()
		return
	end

	local h = c:FindFirstChildOfClass("Humanoid")
	local r = c:FindFirstChild("HumanoidRootPart")
	if not h or not r then
		stopFly()
		return
	end

	-- Ensure fly objects exist
	if not State.FlyBV or State.FlyBV.Parent ~= r then
		startFly()
	end

	if not State.FlyBV or not State.FlyBG then
		return
	end

	local cam = camera()
	if not cam then
		return
	end

	local speed = Config.Movement.FlySpeed

	-- BodyGyro locks orientation to camera
	State.FlyBG.CFrame = cam.CFrame

	local velocity = Vector3.zero

	-- Prefer ControlModule (correct mobile joystick, no inversion)
	local move = Vector3.zero
	if State.FlyControlModule then
		local ok, result = pcall(function()
			return State.FlyControlModule:GetMoveVector()
		end)
		if ok and result then
			move = result
		end
	end

	-- Fallback to Humanoid.MoveDirection if ControlModule unavailable
	if move.Magnitude < 0.01 then
		move = h.MoveDirection
	end

	-- Correct mapping (same as your script – stops inversion)
	if move.X ~= 0 then
		velocity = velocity + cam.CFrame.RightVector * move.X * speed
	end
	if move.Z ~= 0 then
		velocity = velocity - cam.CFrame.LookVector * move.Z * speed
	end

	-- Jump / ButtonA = up, Crouch / ButtonB = down
	if UserInputService:IsKeyDown(Enum.KeyCode.Space)
		or UserInputService:IsKeyDown(Enum.KeyCode.ButtonA) then
		velocity = velocity + Vector3.new(0, speed, 0)
	elseif UserInputService:IsKeyDown(Enum.KeyCode.LeftControl)
		or UserInputService:IsKeyDown(Enum.KeyCode.C)
		or UserInputService:IsKeyDown(Enum.KeyCode.ButtonB) then
		velocity = velocity + Vector3.new(0, -speed, 0)
	end

	State.FlyBV.Velocity = velocity
	h.PlatformStand = true
end

-- CAM LOCK
------------------------------------------------------------

local function updateCamLock(dt)
	-- Main UI toggle enables the feature + FOV.
	-- Floating Lock button only pauses the actual lock via State.CamLockActive.
	if not Config.Combat.CamLock or not State.CamLockActive then
		if not Config.Combat.CamLock then
			State.CamTarget = nil
		end
		return
	end

	local cam = camera()

	if not cam then
		return
	end

	local target = State.CamTarget

	if not target or not validTarget(target) then
		target = getTarget()
		State.CamTarget = target
	end

	if not target then
		return
	end

	local aim = getAimPart(target)

	if not aim then
		State.CamTarget = nil
		return
	end

	-- Smooth look-at selected aim point (Head / Chest / LowerChest)
	local current = cam.CFrame
	local desired = CFrame.lookAt(current.Position, aim.Position)

	-- Higher slider = MORE smooth (slower lock). Lower = snappier.
	-- Map 1..30 → fast..slow
	local smooth = math.clamp(Config.Combat.CamLockSmoothing, 1, 30)
	local speed = (31 - smooth) * 1.2 -- 1 → ~36 (fast), 30 → ~1.2 (very smooth)
	local alpha = 1 - math.exp(-speed * dt)

	local newCF = current:Lerp(desired, alpha)
	cam.CFrame = CFrame.new(current.Position) * (newCF - newCF.Position)
end

local function updateFOV()
	local cam = camera()

	if not cam then
		return
	end

	FOV.Size =
		UDim2.fromOffset(
			Config.Combat.AimFOV * 2,
			Config.Combat.AimFOV * 2
		)

	FOV.Position =
		UDim2.fromOffset(
			cam.ViewportSize.X/2,
			cam.ViewportSize.Y/2
		)

	FOV.Visible =
		State.KeyVerified
		and Config.Combat.CamLock

	fovStroke.Color =
		Config.Settings.FOVColor
end

-- SAVE / LOAD
------------------------------------------------------------

local function copyTable(source)
	local result = {}

	for k,v in pairs(source) do
		if type(v) == "table" then
			result[k] = copyTable(v)
		else
			result[k] = v
		end
	end

	return result
end

Save.Activated:Connect(function()
	local name =
		ConfigName.Text:gsub(
			"^%s*(.-)%s*$",
			"%1"
		)

	if name == "" then
		ConfigStatus.Text = "Enter a config name."
		return
	end

	Configs[name] = copyTable(Config)

	ConfigStatus.Text =
		"Saved: " .. name
end)

Load.Activated:Connect(function()
	local name =
		ConfigName.Text:gsub(
			"^%s*(.-)%s*$",
			"%1"
		)

	if not Configs[name] then
		ConfigStatus.Text = "Config not found."
		return
	end

	local loaded = copyTable(Configs[name])

	for category, values in pairs(loaded) do
		for setting, value in pairs(values) do
			Config[category][setting] = value
		end
	end

	for path, setter in pairs(Controls) do
		local r,k =
			string.match(
				path,
				"^(.-)%.([^%.]+)$"
			)

		if r and k then
			setter(Config[r][k])
		end
	end

	ConfigStatus.Text =
		"Loaded: " .. name
end)

Delete.Activated:Connect(function()
	local name =
		ConfigName.Text:gsub(
			"^%s*(.-)%s*$",
			"%1"
		)

	if Configs[name] then
		Configs[name] = nil
		ConfigStatus.Text =
			"Deleted: " .. name
	else
		ConfigStatus.Text =
			"Config not found."
	end
end)
